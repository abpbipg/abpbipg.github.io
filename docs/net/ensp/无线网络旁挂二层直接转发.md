# 无线网络旁挂二层直接转发

![image-20250923165149168](pi\image-20250923165149168.png)

```
[AC1]
 sysname AC1	
 vlan batch 100
  local-user admin password irreversible-cipher password[]
  
 interface Vlanif100
 ip address 10.23.100.1 255.255.255.0
 dhcp select interface
 
 capwap source interface vlanif100			capwap使用vlanif100为管理vlan
 
 security-profile name wlan-net				安全策略wlan-net
  security wpa-wpa2 psk pass-phrase password[] aes		
  ssid-profile name wlan-net
  ssid wlan-net
  
  vap-profile name wlan-net					wlan-net
  service-vlan vlan-id 101					业务网段为vlan101
  
  ssid-profile wlan-net
  security-profile wlan-net
  
  ap-group name ap-group1
  vap-profile wlan-net wlan 1 radio all		把vap-profile wlan-net wlan 1应用全部raido
  
 ap-id 0 type-id 60 ap-mac 00e0-fc66-7900 ap-sn 210235448310712E346B	绑定ap
  ap-name AP1
  ap-group ap-group1

```

```
[SW2]
interface Vlanif100
 ip address 10.23.100.2 255.255.255.0
#
interface Vlanif101
 ip address 10.23.101.1 255.255.255.0
 dhcp select interface
#
interface MEth0/0/1
#
interface GigabitEthernet0/0/1
 port link-type trunk
 port trunk allow-pass vlan 100 to 101
#
interface GigabitEthernet0/0/2
 port link-type trunk
 port trunk allow-pass vlan 100
#
interface GigabitEthernet0/0/3
 port link-type access
 port default vlan 101

ip route-static 0.0.0.0 0.0.0.0 10.23.101.2
```
